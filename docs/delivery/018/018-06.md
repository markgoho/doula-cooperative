# 018-06 Optimize search performance and lazy loading

[Back to task list](./tasks.md)

## Description

This task focuses on optimizing the performance of the search functionality by implementing lazy loading for Pagefind assets, minimizing initial page load impact, and ensuring efficient resource usage. The goal is to load search assets only when needed while maintaining a smooth user experience.

## Requirements

1. **Lazy Loading Implementation**
   - Load Pagefind assets only when user interacts with search
   - Implement on focus, click, or hover triggers
   - Preload assets on user intent signals
   - Minimize impact on initial page load

2. **Performance Targets**
   - Initial page load unaffected by search
   - Search initialization under 500ms after trigger
   - Total search bundle under 500KB compressed
   - Smooth transition when loading assets

3. **Resource Optimization**
   - Implement proper caching headers
   - Use resource hints for faster loading
   - Optimize chunk loading strategy
   - Monitor and limit memory usage

4. **User Experience**
   - Provide loading indicators
   - Ensure no layout shifts
   - Maintain responsiveness during load
   - Handle slow connections gracefully

## Implementation Plan

1. **Implement lazy loading trigger**

   ```javascript
   (function () {
     let pagefindInitialized = false;
     let initializationPromise = null;

     const searchForm = document.getElementById("search-form");
     const searchInput = searchForm.querySelector('input[type="search"]');
     const enhancedSearch = document.getElementById("pagefind-search");

     // Preload on hover/focus intent
     function preloadPagefind() {
       if (!pagefindInitialized && !initializationPromise) {
         // Add resource hints
         const preconnect = document.createElement("link");
         preconnect.rel = "preconnect";
         preconnect.href = "/pagefind";
         document.head.appendChild(preconnect);

         // Prefetch core assets
         const prefetchJS = document.createElement("link");
         prefetchJS.rel = "prefetch";
         prefetchJS.href = "/pagefind/pagefind-ui.js";
         document.head.appendChild(prefetchJS);

         const prefetchCSS = document.createElement("link");
         prefetchCSS.rel = "prefetch";
         prefetchCSS.href = "/pagefind/pagefind-ui.css";
         document.head.appendChild(prefetchCSS);
       }
     }

     // Initialize Pagefind when needed
     async function initializePagefind() {
       if (pagefindInitialized) return;

       if (!initializationPromise) {
         initializationPromise = loadPagefind();
       }

       return initializationPromise;
     }

     // Load and initialize Pagefind
     async function loadPagefind() {
       try {
         // Show loading state
         searchInput.disabled = true;
         searchInput.placeholder = "Loading search...";

         // Load CSS
         const cssPromise = new Promise((resolve, reject) => {
           const link = document.createElement("link");
           link.rel = "stylesheet";
           link.href = "/pagefind/pagefind-ui.css";
           link.onload = resolve;
           link.onerror = reject;
           document.head.appendChild(link);
         });

         // Load JS
         const jsPromise = new Promise((resolve, reject) => {
           const script = document.createElement("script");
           script.src = "/pagefind/pagefind-ui.js";
           script.onload = resolve;
           script.onerror = reject;
           document.body.appendChild(script);
         });

         // Wait for both to load
         await Promise.all([cssPromise, jsPromise]);

         // Initialize UI
         new PagefindUI({
           element: "#search",
           showSubResults: true,
           showImages: false,
           resetStyles: false,
           autofocus: true,
         });

         // Switch to enhanced search
         searchForm.style.display = "none";
         enhancedSearch.style.display = "block";

         pagefindInitialized = true;

         // Focus the Pagefind input
         const pagefindInput = enhancedSearch.querySelector("input");
         if (pagefindInput) {
           pagefindInput.focus();
         }
       } catch (error) {
         console.error("Failed to initialize Pagefind:", error);
         searchInput.disabled = false;
         searchInput.placeholder = "Search for what's important";
         throw error;
       }
     }

     // Event listeners
     searchInput.addEventListener("mouseenter", preloadPagefind);
     searchInput.addEventListener("focusin", async () => {
       await initializePagefind();
     });

     searchForm.addEventListener("submit", async e => {
       e.preventDefault();
       await initializePagefind();
     });
   })();
   ```

2. **Add performance monitoring**

   ```javascript
   // Track initialization time
   const startTime = performance.now();
   await initializePagefind();
   const loadTime = performance.now() - startTime;

   // Log to analytics if available

     gtag("event", "timing_complete", {
       name: "pagefind_init",
       value: Math.round(loadTime),
     });
   ```

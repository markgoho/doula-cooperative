   - Search indexes all doula profiles, bios, specializations, locations
   - Search UI replaces placeholder form successfully
   - Results display page titles, snippets, and links correctly
   - No server-side infrastructure required
   - Search index generates during build process
   - Performance meets optimization targets
   - Full accessibility compliance

2. **User Journey Testing**
   - Complete flow from search initiation to result selection
   - Test various search scenarios and use cases
   - Validate all user interaction points
   - Ensure smooth experience across devices

3. **Integration Testing**
   - Search works within site context
   - All indexed content is searchable
   - Links navigate correctly
   - No broken functionality

4. **Performance Validation**
   - Initial load time unaffected
   - Search initialization < 500ms
   - Bundle size < 500KB compressed
   - Smooth user experience

## Implementation Plan

1. **Create E2E test scenarios**

   ```javascript
   // Complete user journey tests
   describe("Search Functionality E2E CoS Tests", () => {
     describe("Search Indexing", () => {
       test("indexes all doula profiles", async ({ page }) => {
         await page.goto("/find-a-doula/search");
         await page.focus("#search-input");
         await page.waitForSelector(".pagefind-ui__search-input");

         // Test searching for known doulas
         const testDoulas = ["Jane Doe", "Mary Smith", "Sarah Johnson"];
         for (const doula of testDoulas) {
           await page.fill(".pagefind-ui__search-input", doula);
           await page.waitForSelector(".pagefind-ui__result");

           const results = await page.$$eval(
             ".pagefind-ui__result-title",
             els => els.map(el => el.textContent),
           );
           expect(results.some(r => r.includes(doula))).toBeTruthy();
         }
       });

       test("searches specializations", async ({ page }) => {
         await page.goto("/find-a-doula/search");
         await page.focus("#search-input");
         await page.waitForSelector(".pagefind-ui__search-input");

         await page.fill(".pagefind-ui__search-input", "postpartum");
         await page.waitForSelector(".pagefind-ui__result");

         const results = await page.$$(".pagefind-ui__result");
         expect(results.length).toBeGreaterThan(0);

         // Verify results contain postpartum content
         const excerpts = await page.$$eval(
           ".pagefind-ui__result-excerpt",
           els => els.map(el => el.textContent.toLowerCase()),
         );
         expect(excerpts.some(e => e.includes("postpartum"))).toBeTruthy();
       });
     });

     describe("UI Functionality", () => {
       test("replaces placeholder form with Pagefind UI", async ({ page }) => {
         await page.goto("/find-a-doula/search");

         // Fallback form should be visible initially
         await expect(page.locator("#search-form")).toBeVisible();

         // Focus triggers enhancement
         await page.focus("#search-input");

         // Pagefind UI should replace form
         await expect(page.locator("#search-form")).toBeHidden();
         await expect(page.locator(".pagefind-ui")).toBeVisible();
       });

       test("displays results with title, snippet, and link", async ({
         page,
       }) => {
         await page.goto("/find-a-doula/search");
         await page.focus("#search-input");
         await page.waitForSelector(".pagefind-ui__search-input");

         await page.fill(".pagefind-ui__search-input", "birth doula");
         await page.waitForSelector(".pagefind-ui__result");

         const firstResult = page.locator(".pagefind-ui__result").first();

         // Verify all required elements
         await expect(
           firstResult.locator(".pagefind-ui__result-title"),
         ).toBeVisible();
         await expect(
           firstResult.locator(".pagefind-ui__result-excerpt"),
         ).toBeVisible();
         await expect(
           firstResult.locator(".pagefind-ui__result-link"),
         ).toHaveAttribute("href");
       });
     });

     describe("Performance Requirements", () => {
       test("search initializes within 500ms", async ({ page }) => {
         await page.goto("/find-a-doula/search");

         const startTime = Date.now();
         await page.focus("#search-input");
         await page.waitForSelector(".pagefind-ui__search-input");
         const loadTime = Date.now() - startTime;

         expect(loadTime).toBeLessThan(500);
       });

       test("bundle size is optimized", async ({ page }) => {
         const response = await page.goto("/pagefind/pagefind-ui.js");
         const size = Number(response.headers()["content-length"]);

         // Check compressed size if available
         const encoding = response.headers()["content-encoding"];
         if (encoding === "gzip" || encoding === "br") {
           expect(size).toBeLessThan(500 * 1024); // 500KB
         }
       });
     });

     describe("Accessibility Compliance", () => {
       test("search is keyboard navigable", async ({ page }) => {
         await page.goto("/find-a-doula/search");

         // Tab to search input
         await page.keyboard.press("Tab");
         await expect(page.locator("#search-input")).toBeFocused();

         // Trigger enhancement
         await page.keyboard.press("Enter");
         await page.waitForSelector(".pagefind-ui__search-input");

         // Type search
         await page.keyboard.type("doula");
         await page.waitForSelector(".pagefind-ui__result");

         // Navigate results with keyboard
         await page.keyboard.press("Tab");
         const focusedElement = await page.evaluate(
           () => document.activeElement.className,
         );
         expect(focusedElement).toContain("pagefind-ui__result");
       });

       test("announces results to screen readers", async ({ page }) => {
         // Monitor ARIA live region
         await page.goto("/find-a-doula/search");

         await page.focus("#search-input");
         await page.waitForSelector(".pagefind-ui__search-input");

         // Check for live region
         const liveRegion = await page.$('[aria-live="polite"]');
         expect(liveRegion).toBeTruthy();

         // Search and verify announcement
         await page.fill(".pagefind-ui__search-input", "birth");
         await page.waitForTimeout(1000); // Wait for announcement

         const announcement = await page.$eval(
           '[aria-live="polite"]',
           el => el.textContent,
         );
         expect(announcement).toMatch(/\d+ results? found/);
       });
     });
   });
   ```
